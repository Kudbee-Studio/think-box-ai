## kilo_exec_table.py
"""Gate-function executions in one spine_contract_summary(): the saved baseline profile (main) against a fresh profile of this tree."""
import cProfile, collections, pstats, re, sys
from thinkbox import kilo_live_proof_readiness as spine

GATE = re.compile(r"^(hermetic_.*_check|evaluate_.*|.*_contract_summary|.*_gate_closed)$")

def counts(stats):
    c = collections.Counter(); cum = collections.Counter()
    for (f, l, n), (cc, nc, tt, ct, callers) in stats.stats.items():
        if "/thinkbox/kilo_" in f.replace("\\", "/") and GATE.match(n) and not f.endswith("kilo_eval_scope.py"):
            c[(f.split("/")[-1][:-3], n)] += nc
            cum[(f.split("/")[-1][:-3], n)] += ct
    return c, cum

before, before_cum = counts(pstats.Stats(sys.argv[1]))
pr = cProfile.Profile(); pr.enable(); spine.spine_contract_summary(); pr.disable()
after, _ = counts(pstats.Stats(pr))
print(f"{'gate function':<78}{'before':>9}{'after':>7}")
for key, n in before.most_common(12):
    print(f"{key[0].replace('kilo_', '')[:42] + '.' + key[1]:<78}{n:>9,}{after.get(key, 0):>7}")
print(f"{'all gate functions (' + str(len(before)) + ' distinct before, ' + str(len(after)) + ' after)':<78}{sum(before.values()):>9,}{sum(after.values()):>7}")
print(f"most executions of any one gate function: before {max(before.values()):,}, after {max(after.values())}")

## kilo_count_memo.py
import cProfile, collections, pstats, re, time, io
from thinkbox import kilo_live_proof_readiness as spine
pr = cProfile.Profile(); t = time.time(); pr.enable()
spine.spine_contract_summary()
pr.disable(); wall = time.time() - t
st = pstats.Stats(pr)
GATE = re.compile(r"^(hermetic_.*_check|evaluate_.*|.*_contract_summary|.*_gate_closed)$")
bodies = collections.Counter(); wrappers = 0
for (f, l, n), (cc, nc, tt, ct, callers) in st.stats.items():
    if "/thinkbox/kilo_" in f and GATE.match(n):
        bodies[(f.split("/")[-1], n)] += nc
    if f.endswith("kilo_eval_scope.py") and n == "wrapper": wrappers += nc
print(f"wall under profiler: {wall:.1f}s; total function calls: {st.total_calls:,}")
print(f"gate bodies executed: {sum(bodies.values()):,} for {len(bodies)} distinct gate functions; max for one function: {max(bodies.values())}")
print("most executed gate bodies:", [(n, c) for (m, n), c in bodies.most_common(5)])
print(f"cache-wrapper invocations: {wrappers:,}")
