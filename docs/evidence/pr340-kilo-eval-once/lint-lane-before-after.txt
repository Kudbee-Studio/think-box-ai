## lint_lane.py
"""Run the KILO beyond-lint lane (ruff check, ruff format --check, mypy, bandit over the 38-module scope) in this tree."""
import os, sys
from thinkbox import beyond_kilo_lint as lint
env = {**os.environ, "KILO_BEYOND_KILO_LINT_EXECUTE": "1", "CI": "1"}
results, violations = lint.execute_beyond_kilo_lint_suite(env)
print(f"scope: {len(lint.LINT_SCOPE_REL_PATHS)} modules")
for r in results:
    print(f"{r.tool}: ok={r.ok} skipped={r.skipped} returncode={r.returncode}")
    if not r.ok:
        lines = [l for l in r.detail.splitlines() if l.strip()]
        print(f"  {len(lines)} output lines; first 6:")
        for l in lines[:6]: print("   ", l[:170])
print(f"violations: {len(violations)}")

## lint_counts.sh
#!/usr/bin/env bash
# usage: lint_counts.sh <tree>; counts what each lint-lane tool reports for the 38-module scope in that tree
cd "$1"
SCOPE=$(PYTHONDONTWRITEBYTECODE=1 PYTHONPATH=. python -c "from thinkbox.beyond_kilo_lint import LINT_SCOPE_REL_PATHS as p; print(' '.join(p))")
echo "scope: $(echo $SCOPE | wc -w) modules"
echo "--- ruff check: findings by rule"
ruff check $SCOPE --output-format=concise 2>/dev/null | grep -E '^[^ ]+:[0-9]+:[0-9]+: ' | sed -E 's/^[^ ]+ ([A-Z]+[0-9]+).*/\1/' | sort | uniq -c | sort -rn
echo "ruff check total: $(ruff check $SCOPE --output-format=concise 2>/dev/null | grep -cE '^[^ ]+:[0-9]+:[0-9]+: ')"
echo "--- ruff format --check: $(ruff format --check $SCOPE 2>&1 | tail -1)"
echo "--- mypy --follow-imports=skip: $(mypy --follow-imports=skip $SCOPE 2>&1 | tail -1)"
echo "--- bandit: issues by test id"
bandit -q -r $SCOPE -c pyproject.toml -f json 2>/dev/null | python -c "
import json,sys,collections
d=json.load(sys.stdin); c=collections.Counter(r['test_id']+' '+r['test_name'] for r in d['results'])
for k,v in c.most_common(): print(f'{v:4d} {k}')
print('bandit total:', len(d['results']))"
