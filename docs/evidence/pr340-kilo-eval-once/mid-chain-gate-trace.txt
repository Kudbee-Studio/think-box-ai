"""Trace one mid-chain KILO gate: entry, inputs, children, re-entry, state, termination, fan-out and executions per request.
usage (from a tree): kilo_trace_gate.py <function name> <baseline .prof>"""
import ast, cProfile, collections, pathlib, pstats, re, sys

target, prof = sys.argv[1], sys.argv[2]
GATE = re.compile(r"^(hermetic_.*_check|evaluate_.*|.*_contract_summary|.*_gate_closed)$")
root = pathlib.Path("thinkbox")
defs = {}
for path in sorted(root.glob("kilo_*.py")):
    for node in ast.parse(path.read_text(encoding="utf-8")).body:
        if isinstance(node, ast.FunctionDef) and GATE.match(node.name):
            defs[node.name] = (path, node)
path, node = defs[target]
calls = lambda n: sorted({(c.func.id if isinstance(c.func, ast.Name) else c.func.attr) for c in ast.walk(n) if isinstance(c, ast.Call) and isinstance(c.func, (ast.Name, ast.Attribute))})
children = [c for c in calls(node) if c in defs and c != target]
callers = sorted(k for k, (_, n) in defs.items() if target in calls(n) and k != target)
reads = sorted({ast.unparse(c.func) for c in ast.walk(node) if isinstance(c, ast.Call) and ast.unparse(c.func).endswith(("read_text", "open", "is_file", "exists", "glob", "environ.get", "os.getenv"))})
deco = [ast.unparse(d) for d in node.decorator_list]
print(f"{target}  ({path}:{node.lineno}), decorators {deco}")
print(f"  arguments: {[a.arg for a in node.args.args + node.args.kwonlyargs]}")
print(f"  reads directly: {reads or 'nothing but its arguments'}")
print(f"  gate functions it calls (fan-out {len(children)}): {children}")
print(f"  gate functions that call it (fan-in {len(callers)}): {callers}")
# cycles anywhere in the gate graph (name-based, so conservative)
graph = {k: [c for c in calls(n) if c in defs and c != k] for k, (_, n) in defs.items()}
state, cycle = {}, []
def dfs(u, stack):
    state[u] = 1
    for v in graph[u]:
        if state.get(v) == 1: cycle.append(stack + [u, v]); return
        if v not in state: dfs(v, stack + [u])
        if cycle: return
    state[u] = 2
for k in graph:
    if k not in state: dfs(k, [])
    if cycle: break
print(f"  re-entry/termination: gate call graph has {len(defs)} nodes and {'a CYCLE ' + ' -> '.join(cycle[0]) if cycle else 'no cycle (a DAG: every gate calls only earlier layers)'}")
# measured, one spine_contract_summary()
def stat_of(st, name):
    out = collections.Counter(); by_caller = collections.Counter()
    for (f, l, n), (cc, nc, tt, ct, cs) in st.stats.items():
        if n == name and "/thinkbox/kilo_" in f:
            out["n"] += nc; out["cum"] += ct
            for (cf, cl, cn), v in cs.items(): by_caller[cn] += v[0]
    return out, by_caller
before, bc = stat_of(pstats.Stats(prof), target)
from thinkbox import kilo_live_proof_readiness as spine
pr = cProfile.Profile(); pr.enable(); spine.spine_contract_summary(); pr.disable()
after, ac = stat_of(pstats.Stats(pr), target)
print(f"  executions in one spine_contract_summary(): main {before['n']:,}  ->  this branch {after['n']}   (cumulative on main: {before['cum']:.1f} s)")
print("  who called it on main:", ", ".join(f"{k} x{v:,}" for k, v in bc.most_common(5)))
print("  who called it here:   ", ", ".join(f"{k} x{v}" for k, v in ac.most_common(5)))
