## kilo_prof_save.py
import cProfile, time
from thinkbox import kilo_live_proof_readiness as spine
pr = cProfile.Profile(); t = time.time(); pr.enable()
summary = spine.spine_contract_summary()
pr.disable(); pr.dump_stats("./kilo-spine.prof")
print(f"wall {time.time()-t:.1f}s; summary keys: {len(summary)}")

## kilo_prof_analyze.py
import pstats, re, collections, sys
st = pstats.Stats(sys.argv[1])
rows = []   # (file, line, name, cc, nc, tt, ct)
for (f, l, n), (cc, nc, tt, ct, callers) in st.stats.items():
    rows.append((f, l, n, cc, nc, tt, ct))
total_self = sum(r[5] for r in rows)
def cat(f, n):
    if "sqlite3" in n or "sqlite" in f: return "persistence: SQLite (execute/executescript/connect)"
    if "'search' of 're.Pattern'" in n or "'match' of 're.Pattern'" in n or "'sub' of 're.Pattern'" in n or "'finditer'" in n or "'findall'" in n or f.endswith("/re/__init__.py") or "/re/_" in f or f.endswith("re.py") or "sre_" in f or "_compile" in n and "re" in f:
        return "documentation scanning: regex over doc text"
    if "/json/" in f or "json" in n.lower() and "built-in" in n: return "serialization: json encode/decode"
    if "<frozen os>" in f or "os.environ" in n: return "environment: os.environ lookups"
    if "io.open" in n or "posix." in n or "pathlib" in f or "genericpath" in f or "posixpath" in f or "io.TextIOWrapper" in n or "_io." in n or "FileIO" in n:
        return "filesystem I/O (open/read/stat/path handling)"
    if "subprocess" in f or "subprocess" in n: return "subprocesses"
    if "socket" in f or "urllib" in f or "http" in f or "ssl" in f: return "network"
    if "/thinkbox/kilo_" in f: return "gate logic: thinkbox/kilo_*.py own code"
    if "/thinkbox/" in f: return "support modules: thinkbox/* (non-gate)"
    return "interpreter/builtins (isinstance, len, dict/list ops, ...)"
bycat = collections.Counter()
for f, l, n, cc, nc, tt, ct in rows: bycat[cat(f, n)] += tt
print(f"total self-time: {total_self:.1f}s ({st.total_calls:,} calls)\n")
print("SELF-TIME BY CATEGORY")
for c, t in bycat.most_common(): print(f"  {t:7.1f}s {100*t/total_self:5.1f}%  {c}")
print("\nTOP 12 FUNCTIONS BY SELF-TIME")
for f, l, n, cc, nc, tt, ct in sorted(rows, key=lambda r: -r[5])[:12]:
    short = f.split("/site-packages/")[-1].split("/think-box-ai/")[-1].split("/lib/python3.11/")[-1]
    print(f"  {tt:6.1f}s {100*tt/total_self:5.1f}%  {nc:>9,} calls  {short}:{l}({n})")
# gate functions
GATE = re.compile(r"^(hermetic_.*_check|evaluate_.*|.*_contract_summary|.*_gate_closed)$")
gates = [(f, n, nc, tt, ct) for f, l, n, cc, nc, tt, ct in rows if "/thinkbox/kilo_" in f and GATE.match(n)]
ev = [g for g in gates if g[1].startswith("evaluate_")]
print(f"\nGATE FUNCTIONS EXECUTED: {len(gates)} distinct, {sum(g[2] for g in gates):,} executions in one spine_contract_summary()")
print(f"  evaluate_* : {len(ev)} distinct, {sum(g[2] for g in ev):,} executions -> {sum(g[2] for g in ev)/max(1,len(ev)):.0f} per gate on average")
print("  most re-evaluated evaluate_* functions:")
for f, n, nc, tt, ct in sorted(ev, key=lambda g: -g[2])[:6]:
    print(f"    {nc:>7,} x  {n}   (own {tt/nc*1000:.2f} ms/call, cumulative {ct/nc*1000:.1f} ms/call)")
# estimated work if each distinct evaluate ran once = sum over distinct of (cumulative per call) is double counting; use own time per call x1 plus callee categories unknown -> use cumulative of leaf-most only
dist_cost = sum(g[4]/g[2] for g in ev if g[2] > 0)
print(f"  sum of per-call cumulative cost over the {len(ev)} distinct evaluate_* functions (upper bound for one pass, double counts nesting): {dist_cost:.1f}s")
