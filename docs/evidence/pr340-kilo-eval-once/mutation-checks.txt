def run(test):
    r = subprocess.run([sys.executable, "-m", "pytest", test, "-q", "-x", "--no-header", "-p", "no:cacheprovider", "--timeout=120", "--tb=no", "-rf"],
                       capture_output=True, text=True, env={**__import__("os").environ, "PYTHONDONTWRITEBYTECODE": "1", "PYTHONPATH": str(ROOT)})
    failed = re.findall(r"^FAILED (\S+)", r.stdout, re.M)
    return r.returncode, failed, r.stdout.strip().splitlines()[-1] if r.stdout.strip() else ""
