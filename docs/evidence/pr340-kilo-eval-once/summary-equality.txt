import json, re, collections, sys
def flat(v, prefix=""):
    if isinstance(v, dict):
        for k in sorted(v): yield from flat(v[k], f"{prefix}.{k}" if prefix else k)
    elif isinstance(v, list):
        if not v: yield prefix, []
        for i, x in enumerate(v): yield from flat(x, f"{prefix}[{i}]")
    else: yield prefix, v
a = dict(flat(json.load(open(sys.argv[1])))); b = dict(flat(json.load(open(sys.argv[2]))))
diff = [(k, a.get(k, "<absent>"), b.get(k, "<absent>")) for k in sorted(set(a) | set(b)) if a.get(k, "<absent>") != b.get(k, "<absent>")]
cat = collections.defaultdict(list)
for k, x, y in diff:
    leaf = k.split(".")[-1]
    if leaf in ("gate_closed_default", "hermetic_operator_ok"): cat["verdict flips False -> True (gate_closed_default, hermetic_operator_ok)"].append((k, x, y))
    elif leaf.startswith("hermetic_violation_codes"): cat["a violation code disappears (hermetic_violation_codes)"].append((k, x, y))
    elif x is False and y is True and leaf.endswith(("_ok", "_layer_ok")): cat["a prior-layer flag flips False -> True (*_ok)"].append((k, x, y))
    else: cat["listed individually below"].append((k, x, y))
print(f"{len(a)} leaf values on main, {len(b)} on this branch; {len(diff)} differ:")
for c, items in cat.items(): print(f"  {len(items):4d}  {c}")
print("values that go from True to False:", [k for k, x, y in diff if x is True and y is False] or "none")
print("gates whose verdict flips:", len({k.split('.')[0] for k, _, _ in cat['verdict flips False -> True (gate_closed_default, hermetic_operator_ok)']}))
print("\nlisted individually:")
for k, x, y in cat["listed individually below"]: print(f"  {k}: {x!r} -> {y!r}")
